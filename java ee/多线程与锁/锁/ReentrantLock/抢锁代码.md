```java
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.locks.ReentrantLock;

public class LockCompetitionDemo {
    private final ReentrantLock lock = new ReentrantLock();
    private int counter = 0;  // 共享资源

    public static void main(String[] args) {
        LockCompetitionDemo demo = new LockCompetitionDemo();
        // 创建5个线程的线程池
        ExecutorService pool = Executors.newFixedThreadPool(5);
        
        // 提交10个任务到线程池
        for (int i = 0; i < 10; i++) {
            pool.submit(() -> demo.tryLockAndPrint());
        }
        
        // 关闭线程池
        pool.shutdown();
    }

    private void tryLockAndPrint() {
        String threadName = Thread.currentThread().getName();
        
        if (lock.tryLock()) {
            try {
                System.out.println(threadName + " ✅ 成功获取锁，当前计数: " + counter);
                // 模拟临界区操作
                counter++;
                Thread.sleep(200);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                System.out.println(threadName + " ❗ 执行过程中被中断");
            } finally {
                lock.unlock();
                System.out.println(threadName + " 🔓 释放锁，新计数: " + counter);
            }
        } else {
            System.out.println(threadName + " ❌ 未获取到锁，执行备选操作");
            // 执行不需要锁的操作
        }
    }
}

```


# 解释：
Trylock（）
Reentrantlock 的 trylock 是非阻塞锁，抢不到直接放弃
